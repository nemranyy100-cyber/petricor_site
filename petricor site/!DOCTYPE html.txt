  script
    document.addEventListener('DOMContentLoaded', () = {
       1. Initialize Lucide Icons
      lucide.createIcons();

       Dynamic Footer Year
      document.getElementById('current-year').textContent = new Date().getFullYear();

       2. Mobile Menu Toggle
      const mobileMenuBtn = document.getElementById('mobile-menu-btn');
      const mobileMenu = document.getElementById('mobile-menu');
      const iconOpen = document.getElementById('menu-icon-open');
      const iconClose = document.getElementById('menu-icon-close');
      const mobileNavLinks = document.querySelectorAll('.mobile-nav-link');

      function toggleMobileMenu(forceClose = false) {
        const isHidden = mobileMenu.classList.contains('hidden');
        if (forceClose  !isHidden) {
          mobileMenu.classList.add('hidden');
          iconOpen.classList.remove('hidden');
          iconClose.classList.add('hidden');
          mobileMenuBtn.setAttribute('aria-expanded', 'false');
        } else {
          mobileMenu.classList.remove('hidden');
          iconOpen.classList.add('hidden');
          iconClose.classList.remove('hidden');
          mobileMenuBtn.setAttribute('aria-expanded', 'true');
        }
      }

      mobileMenuBtn.addEventListener('click', () = toggleMobileMenu());
      mobileNavLinks.forEach(link = {
        link.addEventListener('click', () = toggleMobileMenu(true));
      });

       3. FAQ Accordion
      const faqItems = document.querySelectorAll('.faq-item');
      faqItems.forEach(item = {
        const trigger = item.querySelector('.faq-trigger');
        trigger.addEventListener('click', () = {
          const isCurrentlyActive = item.classList.contains('active');
           Close all items
          faqItems.forEach(otherItem = otherItem.classList.remove('active'));
           Re-open if it wasn't active
          if (!isCurrentlyActive) {
            item.classList.add('active');
          }
        });
      });

       4. Registration Form - Pre-filled WhatsApp Message + Confirmation UI
      const bootcampForm = document.getElementById('bootcamp-form');
      const confirmationBox = document.getElementById('form-confirmation');
      const whatsappFallbackLink = document.getElementById('whatsapp-fallback-link');
      const resetFormBtn = document.getElementById('reset-form-btn');

       =========================================================================
       CONFIGURATION WHATSAPP  Remplacez le numéro ci-dessous (sans + ni espaces)
       par votre numéro WhatsApp Business ou personnel (format international).
       Exemple Maroc  212612345678 pour +212 6 12 34 56 78
       =========================================================================
      const ACADEMY_WHATSAPP_NUMBER = '212600000000';

      bootcampForm.addEventListener('submit', (e) = {
        e.preventDefault();

        const fullName = document.getElementById('fullName').value.trim();
        const socialHandle = document.getElementById('socialHandle').value.trim();
        const whatsappNumber = document.getElementById('whatsappNumber').value.trim();
        const tradingLevel = document.getElementById('tradingLevel').value;

        if (!fullName  !socialHandle  !whatsappNumber  !tradingLevel) {
          return;
        }

        const formattedMessage =
          `Bonjour Petricor Académie,nn` +
          `Je souhaite réserver ma place pour le prochain Bootcamp de Trading. Voici mes informations nn` +
          `• Nom Complet  ${fullName}n` +
          `• Pseudo Discord  Instagram  ${socialHandle}n` +
          `• Numéro WhatsApp  ${whatsappNumber}n` +
          `• Niveau actuel  ${tradingLevel}nn` +
          `Merci de m'indiquer les prochaines étapes pour rejoindre le Discord privé !`;

        const whatsappUrl = `httpswa.me${ACADEMY_WHATSAPP_NUMBER}text=${encodeURIComponent(formattedMessage)}`;

         Update fallback button URL & display confirmation message on screen
        whatsappFallbackLink.href = whatsappUrl;
        bootcampForm.classList.add('hidden');
        confirmationBox.classList.remove('hidden');

         Open pre-filled WhatsApp message in a new tab
        window.open(whatsappUrl, '_blank', 'noopener,noreferrer');
      });

      resetFormBtn.addEventListener('click', () = {
        confirmationBox.classList.add('hidden');
        bootcampForm.classList.remove('hidden');
      });
    });
